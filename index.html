async function calcularReserva(e) { e.preventDefault(); const apt = document.getElementById('apt-select').value; const checkinVal = document.getElementById('checkin').value; const checkoutVal = document.getElementById('checkout').value; const hospedes = parseInt(document.getElementById('hospedes').value) || 1; if (!checkinVal || !checkoutVal) { alert('Por favor selecione as datas de check-in e check-out.'); return; } const checkin = new Date(checkinVal); const checkout = new Date(checkoutVal); const diffTime = checkout - checkin; const noites = Math.ceil(diffTime / (1000 * 60 * 60 * 24)); if (noites <= 0) { alert('A data de check-out tem de ser posterior ao check-in.'); return; } // Buscar mapa de preços da API let mapaPrecos = {}; try { const res = await fetch('/api/precos'); const json = await res.json(); if (json.success) mapaPrecos = json.data; } catch(err) {} const precoBaseDefault = apt === 'cristal_mar' ? 65 : 60; let subtotalNoites = 0; let temBloqueio = false; let dt = new Date(checkin); for (let i = 0; i < noites; i++) { const dataStr = dt.toISOString().split('T')[0]; const dadosDia = mapaPrecos[apt] && mapaPrecos[apt][dataStr]; if (dadosDia && dadosDia.bloqueado) { temBloqueio = true; break; } const precoDia = (dadosDia && dadosDia.preco) ? dadosDia.preco : precoBaseDefault; subtotalNoites += precoDia; dt.setDate(dt.getDate() + 1); } if (temBloqueio) { alert('Infelizmente algumas das datas selecionadas não estão disponíveis para reserva.'); return; } const desconto = subtotalNoites * 0.10; const noitesComDesconto = subtotalNoites - desconto; const taxaLimpeza = 35.00; const noitesTaxa = Math.min(noites, 7); const mesCheckin = checkin.getMonth(); const eEpocaAlta = (mesCheckin >= 5 && mesCheckin <= 8); const valorTaxaPorNoite = eEpocaAlta ? 2.00 : 1.50; const taxaTuristica = noitesTaxa * hospedes * valorTaxaPorNoite; const totalFinal = noitesComDesconto + taxaLimpeza + taxaTuristica; document.getElementById('label-noites').innerText = `Estadia (${noites} noites):`; document.getElementById('valor-noites').innerText = `${subtotalNoites.toFixed(2)} €`; document.getElementById('valor-desconto').innerText = `-${desconto.toFixed(2)} €`; document.getElementById('valor-limpeza').innerText = `${taxaLimpeza.toFixed(2)} €`; document.getElementById('label-taxa-turistica').innerText = `Taxa Turística F. Foz (${noitesTaxa} noites x ${hospedes} hósp. x ${valorTaxaPorNoite.toFixed(2)}€ - Isenta):`; document.getElementById('valor-taxa-turistica').innerText = `${taxaTuristica.toFixed(2)} €`; document.getElementById('valor-total').innerText = `${totalFinal.toFixed(2)} €`; document.getElementById('resultado-simulacao').classList.remove('hidden'); document.getElementById('modal-ifthenpay').classList.add('hidden'); document.getElementById('resultado-simulacao').scrollIntoView({ behavior: 'smooth' }); }
